Memo Brief

this memo explore two main ideas about the automated camera enforcement in Washington, D.C.

1- Revenue v. safety: how much money the city actually billed in traffic enforcement and are they perioratizing safety or revenue.

2- Racial distribution: are wards with majority black population more likely to experience more automated enforcement percapita.

Data sources:

  • MPD arrest data for 2024

  • DC automated traffic enforcement locations

  • DC automated camera violations by month (2024)

  • ACS 5-year ward demographics

  • 2024 moving violations

ward_pop <- demographics |> 
  mutate(ward_num = str_extract(namelsad, "\\d")) |> 
  select(ward_num, 
         total_pop = dp05_0001e,

  mutate(ward_num = str_extract(namelsad, "\\d")) |> 
  select(ward_num, 
         total_households = dp02_0001e,

         bachelors_plus = dp02_0068e)

# creating a table for each ward with total arrests and numbers for each racein this one I have to map areests psa to wards so it can match each ward in the district
ward_arrest_stats <- arrests |> 
  mutate(psa = as.character(offense_location_psa)) |> 
  mutate(ward_num = case_when(
    psa %in% c("301", "302", "303", "304", "305", "408") ~ "1",
    psa %in% c("101", "102", "207", "208", "209", "306", "307") ~ "2",
    psa %in% c("201", "202", "203", "204", "205", "206") ~ "3",
    psa %in% c("401", "402", "403", "404", "405", "406", "407", "409") ~ "4",
    psa %in% c("501", "502", "503", "504", "505", "506", "507") ~ "5",
    psa %in% c("103", "104", "105", "106", "107", "108") ~ "6",
    psa %in% c("601", "602", "603", "604", "605", "606", "607", "608") ~ "7",
    psa %in% c("701", "702", "703", "704", "705", "706", "707", "708") ~ "8",
    TRUE ~ NA_character_
  )) |> 
  group_by(ward_num) |> 
summarise(
    total_arrests = n(),
    black_arrests = sum(defendant_race == "BLACK", na.rm = TRUE),
    white_arrests = sum(defendant_race == "WHITE", na.rm = TRUE),
    total_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated"), na.rm = TRUE), 
    black_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated") & defendant_race == "BLACK", na.rm = TRUE), 
    white_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated") & defendant_race == "WHITE", na.rm = TRUE)
  )
#Traffic cameras violations
total_violations <- camera_violations |> 
  group_by(enforcement_space_code) |> 
  summarise(total_violations = sum(num_violations, na.rm = TRUE))

ward_camera_stats <- camera_locs |> 
  mutate(ward_num = as.character(as.integer(ward))) |>

  group_by(ward_num) |> 
  summarise(
    camera_count = n(),

wards <- st_read("data/Wards_dc.geojson", quiet = TRUE) |>

Traffic violation volume by neighborhood

library(sf)
library(tidyverse)
library(lubridate)

# Violation data for 2024
top_locations <- camera_violations |> 
  mutate(date = ymd_hms(year_month)) |> 
  filter(year(date) %in% c(2024)) |> 
  group_by(enforcement_space_code) |> 
  summarise(total_violations = sum(num_violations, na.rm = TRUE))

# perparing camera coordinates to add them to the map
camera_locs_clean <- camera_locs |>
  rename_with(tolower) |>
  rename(
    longitude = any_of(c("longitude","long","x","lon")),
    latitude  = any_of(c("latitude","lat","y"))
  )

# converting camera location to special points
camera_dots_sf <- camera_locs_clean |>
  drop_na(longitude, latitude) |> 
  st_as_sf(coords = c("longitude", "latitude"), crs = 3857) |>
  st_transform(4326)

# Loading DC neighborhood map
neighborhood_sf <- st_read("data/Neighborhood_dc.geojson", quiet = TRUE) |> 
  st_transform(4326)
neighborhood_sf <- neighborhood_sf |> 
  rename_with(tolower) |> 
  rename(neigh_name = any_of(c("name")))

# linkimg violation total and cam locations
camera_points_with_violations <- camera_dots_sf |> 
  left_join(top_locations, by = "enforcement_space_code")
camera_neighborhood <- st_join(camera_points_with_violations, neighborhood_sf)

# calculating total violation in each neighborhood
neighborhood_revenue <- camera_neighborhood |> 
  filter(!is.na(neigh_name)) |> 
  st_drop_geometry() |> 
  group_by(neigh_name) |> 
  summarise(
    total_revenue_vol = sum(total_violations, na.rm = TRUE)
  )
map_data_neighborhood <- neighborhood_sf |> 
  left_join(neighborhood_revenue, by = "neigh_name")

# drawing the map
ggplot() +
  geom_sf(data = map_data_neighborhood,
          aes(fill = total_revenue_vol),
          color = "white",
          linewidth = 0.2) +
  geom_sf(data = camera_dots_sf,
          color = "orange",
          size = 0.4,
          alpha = 0.4) +
  
  # shading
  scale_fill_viridis_c(
    option = "mako",
    direction = -1,
    name = "Ticket Volume",
    labels = scales::comma,
    na.value = "grey90"
  ) +
  
  theme_void() +
    labs(
    title = "Ticket volume by neighberhood",
    caption = "Source: Open Data DC"
  )
