ward_arrest_stats <- arrests |> 
  mutate(psa = as.character(offense_location_psa)) |> 
  mutate(ward_num = case_when(
    psa %in% c("301", "302", "303", "304", "305", "408") ~ "1",
    psa %in% c("101", "102", "207", "208", "209", "306", "307") ~ "2",
    psa %in% c("201", "202", "203", "204", "205", "206") ~ "3",
    psa %in% c("401", "402", "403", "404", "405", "406", "407", "409") ~ "4",
    psa %in% c("501", "502", "503", "504", "505", "506", "507") ~ "5",
    psa %in% c("103", "104", "105", "106", "107", "108") ~ "6",
    psa %in% c("601", "602", "603", "604", "605", "606", "607", "608") ~ "7",
    psa %in% c("701", "702", "703", "704", "705", "706", "707", "708") ~ "8",
    TRUE ~ NA_character_
  )) |> 
  group_by(ward_num) |> 
summarise(
    total_arrests = n(),
    black_arrests = sum(defendant_race == "BLACK", na.rm = TRUE),
    white_arrests = sum(defendant_race == "WHITE", na.rm = TRUE),
    total_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated"), na.rm = TRUE), 
    black_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated") & defendant_race == "BLACK", na.rm = TRUE), 
    white_traffic_arrests = sum(arrest_category %in% c("Traffic Violations", "Driving/Boating While Intoxicated") & defendant_race == "WHITE", na.rm = TRUE)
  )

joined_df <- ward_arrest_stats |> 
  drop_na(ward_num) |> 
  inner_join(ward_pop, by = "ward_num") |> 
  left_join(ward_camera_stats, by = "ward_num") |> 
  mutate(
    # Arrests per 1,000 residents of each race (per capita)
    black_per_capita = (black_arrests / black_pop) * 1000,
    white_per_capita = (white_arrests / white_pop) * 1000,
    #Cameras per 10,000 residents
    cameras_per_10k = (camera_count / total_pop) * 10000,
    # Percentages for dataviz
  pct_black_pop = (black_pop / total_pop) * 100,
    pct_white_pop = (white_pop / total_pop) * 100,
    pct_black_arrests = (black_arrests / total_arrests) * 100,
    pct_white_arrests = (white_arrests / total_arrests) * 100,
    pct_black_traffic_arrests = (black_traffic_arrests / total_traffic_arrests) * 100,

  )

Comparing the traffic enforcement burden from physical arrests and automated cameras in ward 3.5 and 8

# here Iwill get camera violations for 2024
camera_violations_2024 <- camera_violations |> 
  mutate(year_month = ymd_hms(year_month)) |> 
  filter(year(year_month) == 2024)
# summing violations for each camera location
total_violations_2024 <- camera_violations_2024 |> 
  group_by(enforcement_space_code) |> 
  summarise(total_violations_2024 = sum(num_violations, na.rm = TRUE))
# here i will link each camera to its ward so I can calculate totals per ward
ward_camera_stats_2024 <- camera_locs |> 
  mutate(ward_num = as.character(as.integer(ward))) |> 
  left_join(total_violations_2024, by = "enforcement_space_code") |> 
  group_by(ward_num) |> 
  summarise(
    total_camera_violations_2024 = sum(total_violations_2024, na.rm = TRUE)
  ) |> 
  filter(ward_num != "0")
# joining traffic arrests with camera violations 
burden_df <- ward_arrest_stats |> 
  drop_na(ward_num) |> 
  left_join(ward_camera_stats_2024, by = "ward_num")
# calculate the city averages so i can compare each ward to DC overall
avg_fines_2024 <- mean(burden_df$total_camera_violations_2024, na.rm = TRUE)
avg_traffic_arrests <- mean(burden_df$total_traffic_arrests, na.rm = TRUE)
# compare ward 3 5 and 8 enforcement burden
burden_df |> 
  filter(ward_num %in% c("3","5","8")) |> 
  mutate(
    fine_burden = total_camera_violations_2024 / avg_fines_2024,
    arrest_burden = total_traffic_arrests / avg_traffic_arrests,
    Location = paste("Ward", ward_num)
  ) |> 
  select(Location, fine_burden, arrest_burden) |> 
  pivot_longer(cols = -Location,
               names_to = "Type",
               values_to = "Score") |> 
  ggplot(aes(x = Type, y = Score, fill = Location)) +
  geom_col(position = "dodge") +
# adding a line to represent the avrg enforcement experience
    geom_hline(yintercept = 1,
             linetype = "dashed",
             color = "red") +
  
  scale_fill_manual(values = c(
    "Ward 3" = "lightgrey",
    "Ward 5" = "steelblue",
    "Ward 8" = "darkblue"
  )) +
  
  scale_x_discrete(labels = c(
    fine_burden = "Camera Violations (2024)",
    arrest_burden = "Traffic Arrests (2024)"
  )) +
  
  theme_minimal() +
  labs(
    title = "The Traffic Enforcement Burden (2024)",
    subtitle = "Red line = City Average",
    y = "Times Higher Than City Average",
